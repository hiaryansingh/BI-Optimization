Red Sales = CALCULATE ( [Sales], FILTER ( 'Product', 'Product'[Color] = "Red" ) )

Sales YoY Growth % =
DIVIDE (
    [Sales] - CALCULATE ( [Sales], PARALLELPERIOD ( 'Date'[Date], -12, MONTH ) ),
    CALCULATE ( [Sales], PARALLELPERIOD ( 'Date'[Date], -12, MONTH ) )
)

Australian Sales Tax =
IF ( HASONEVALUE ( Customer[Country-Region] ),
    IF ( VALUES ( Customer[Country-Region] ) = "Australia", [Sales] * 0.10 ) )

Sales Orders = COUNT ( Sales[OrderDate] )

Profit Margin = DIVIDE ( [Profit], [Sales], 0 )

Margin per Unit = [Profit] / [Units]

High Value Rev =
SUMX ( FILTER ( 'Sales', 'Sales'[Amount] > 1000 ), 'Sales'[Amount] * 'Sales'[FX Rate] )

Sales for Profitable Months =
CALCULATE ( [Sales], FILTER ( VALUES ( 'Date'[Month] ), [Profit] > 0 ) )

Total Sales = SUM ( Sales[Amount] )
